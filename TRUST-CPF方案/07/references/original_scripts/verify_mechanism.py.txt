def ridge(xs,ys,types,alpha,relative=False):
    x=np.concatenate(xs).astype(float);y=np.concatenate(ys).astype(float);t=np.tile(types,len(xs))
    center=x.mean(0);spread=np.maximum(x.std(0),1e-6)
    scales=np.array([max(y[t==k].std(),1e-12) for k in range(3)])
    z=np.column_stack((np.ones(len(x)),(x-center)/spread))
    z=np.concatenate([z*(t==k)[:,None] for k in range(3)],1);q=y/scales[t]
    if relative:
        # Independent block projection, and equivalence to all-pairs differences.
        for j in range(len(xs)):
            for k in range(3):
                ids=np.flatnonzero(t[j*96:(j+1)*96]==k)+j*96
                p=np.eye(len(ids))-np.ones((len(ids),len(ids)))/len(ids)
                z[ids]=p@z[ids];q[ids]=p@q[ids]
        # Exact zero of projected intercepts, avoid roundoff at the 1e-16 level.
        z[:,[0,9,18]]=0
    beta=np.linalg.solve(z.T@z+alpha*np.eye(27),z.T@q)
    return dict(beta=beta,center=center,spread=spread,scales=scales)
