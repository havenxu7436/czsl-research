def sha(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for b in iter(lambda:f.read(4*1024*1024),b''):h.update(b)
    return h.hexdigest()
def corr(x,y):
    x,y=average_rank(x),average_rank(y)
    return float(np.corrcoef(x,y)[0,1]) if x.std()>0 and y.std()>0 else 0.
def stat(x):
    a=np.asarray(x,dtype=float)
    return dict(values=a.tolist(),mean=float(a.mean()),sd=float(a.std(ddof=1)) if len(a)>1 else None)
