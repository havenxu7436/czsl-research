const {chromium}=require('C:/Users/llenovo/.cache/codex-runtimes/codex-primary-runtime/dependencies/node/node_modules/playwright');
const fs=require('fs');
const path=require('path');
const {pathToFileURL}=require('url');
(async()=>{
 const qa=path.resolve('tmp/g0_three_report_qa');
 const browser=await chromium.launch({executablePath:'C:/Program Files (x86)/Microsoft/Edge/Application/msedge.exe',headless:true});
 try{
  const page=await browser.newPage({viewport:{width:1440,height:1000},deviceScaleFactor:1});
  await page.goto(pathToFileURL(path.join(qa,'preview.html')).href);
  const tables=page.locator('table');
  await tables.nth(3).screenshot({path:path.join(qa,'auc_table.png')});
  await tables.nth(4).screenshot({path:path.join(qa,'hm_table.png')});
  const dims=await page.evaluate(()=>({viewport:innerWidth,documentWidth:document.documentElement.scrollWidth,tables:[...document.querySelectorAll('table')].map((e,i)=>({i,width:e.getBoundingClientRect().width,right:e.getBoundingClientRect().right}))}));
  fs.writeFileSync(path.join(qa,'layout.json'),JSON.stringify(dims,null,2));
  console.log(JSON.stringify({tables:dims.tables.length,viewport:dims.viewport,documentWidth:dims.documentWidth,maxTableWidth:Math.max(...dims.tables.map(t=>t.width))}));
 }finally{await browser.close();}
})().catch(e=>{console.error(e);process.exitCode=1});
