"""Verify authorized stage-11 LFS objects without storing credentials or signed URLs."""
from pathlib import Path
from urllib.parse import urlparse
import argparse,subprocess,os,json,hashlib,time
import requests
R=Path('D:/CodexWorkspaces/trust-g0-three-11-20261101');B=Path('C:/Users/llenovo/Documents/ChatGPT/USG');S=R/'TRUST-CPF方案/11'
def credentials():
 env=dict(os.environ,GIT_TERMINAL_PROMPT='0',GCM_INTERACTIVE='never')
 data=subprocess.check_output(['git','-c','http.proxy=','credential','fill'],cwd=R,input=b'protocol=https\nhost=github.com\n\n',env=env,stderr=subprocess.DEVNULL)
 values=dict(x.split('=',1) for x in data.decode().splitlines() if '=' in x)
 return values['username'],values['password']
def main():
 p=argparse.ArgumentParser();p.add_argument('--out',required=True);p.add_argument('--samples',action='store_true');a=p.parse_args()
 objects=json.loads((B/'tmp/g0_three_lfs_objects.json').read_text(encoding='utf8'))
 client=requests.Session();client.trust_env=False;client.auth=credentials();client.headers.update({'Accept':'application/vnd.git-lfs+json','Content-Type':'application/vnd.git-lfs+json'})
 checked=[]
 for start in range(0,len(objects),100):
  response=client.post('https://github.com/havenxu7436/czsl-research.git/info/lfs/objects/batch',json=dict(operation='upload',transfers=['basic'],objects=objects[start:start+100]),timeout=60)
  assert response.status_code==200,('LFS batch status',response.status_code)
  for obj in response.json()['objects']:
   state='error' if 'error' in obj else 'upload_required' if obj.get('actions',{}).get('upload') else 'present'
   checked.append(dict(oid=obj['oid'],size=obj['size'],state=state))
 result=dict(date='2026-11-01',objects=len(checked),present=sum(r['state']=='present' for r in checked),present_bytes=sum(r['size'] for r in checked if r['state']=='present'),total_bytes=sum(r['size'] for r in checked),results=checked)
 print('REMOTE_PRESENT',result['present'],'/',result['objects'],'BYTES',result['present_bytes'],'/',result['total_bytes'],flush=True)
 if a.samples:
  assert result['present']==result['objects']
  records=json.loads((S/'file_manifest.json').read_text(encoding='utf8'))
  chosen=[]
  for token in ['results/deployment/trustg0_three_v1_10ep.zip','results/archives/cgqa/trustg0_cgqa_results.tar.gz.part000','results/raw_export/trustg0_ut_results/work/ut-zap50k/seed5351/formal/g0/best.pt','results/raw_export/trustg0_mit_results/work/mit-states/seed5352/recognition/context/1701/test_ow.npz','results/raw_export/trustg0_cgqa_results/work/cgqa/seed5353/teachers/t1_q5/measure/response.pt']:
   row=next(r for r in records if r['path']==token);chosen.append(row)
  response=client.post('https://github.com/havenxu7436/czsl-research.git/info/lfs/objects/batch',json=dict(operation='download',transfers=['basic'],objects=[dict(oid=r['sha256'],size=r['bytes']) for r in chosen]),timeout=60)
  assert response.status_code==200
  samples=[];lookup={r['sha256']:r for r in chosen}
  for obj in response.json()['objects']:
   assert 'error' not in obj
   action=obj['actions']['download'];url=action['href'];assert urlparse(url).scheme=='https'
   session=requests.Session();session.trust_env=False
   response=session.get(url,headers=action.get('header',{}),stream=True,timeout=(30,120));assert response.status_code==200,('Download',response.status_code)
   h=hashlib.sha256();size=0
   for chunk in response.iter_content(8*1024**2):h.update(chunk);size+=len(chunk)
   assert h.hexdigest()==obj['oid'] and size==obj['size']
   samples.append(dict(path=lookup[obj['oid']]['path'],bytes=size,sha256=h.hexdigest(),verified=True))
   print('FRESH_DOWNLOAD_OK',lookup[obj['oid']]['path'],size,flush=True)
  result['fresh_downloads']=samples
 Path(a.out).write_text(json.dumps(result,indent=2)+'\n',encoding='utf8')
if __name__=='__main__':
 try:main()
 except requests.RequestException as exc:raise SystemExit('NETWORK_ERROR: '+type(exc).__name__) from None
