const fs=require('fs'),path=require('path'),katex=require('./katex/katex.min.cjs');
const root='D:/CodexWorkspaces/trust-g0-three-11-20261101',stage=path.join(root,'TRUST-CPF方案/11'),fence=String.fromCharCode(96).repeat(3),tick=String.fromCharCode(96);
function all(p){return fs.readdirSync(p,{withFileTypes:true}).flatMap(x=>x.isDirectory()?all(path.join(p,x.name)):[path.join(p,x.name)]);}
const errors=[],results=[];
for(const file of all(stage).filter(p=>p.endsWith('.md'))){
 const source=fs.readFileSync(file,'utf8');let display=0,inline=0;
 for(const m of source.matchAll(new RegExp(fence+'math\\r?\\n([\\s\\S]*?)'+fence,'g'))){
  display++;try{katex.renderToString(m[1],{displayMode:true,throwOnError:true,strict:'ignore'});}catch(e){errors.push({file,error:e.message});}
 }
 for(const m of source.matchAll(new RegExp('\\$'+tick+'([^\\n]*?)'+tick+'\\$','g'))){
  inline++;try{katex.renderToString(m[1],{throwOnError:true,strict:'ignore'});}catch(e){errors.push({file,error:e.message});}
 }
 if(source.split(fence).length%2!==1)errors.push({file,error:'Unclosed fence'});
 let columns=null,inFence=false,tables=0;
 for(const [i,line] of source.split(/\r?\n/).entries()){
  if(line.startsWith(fence)){inFence=!inFence;columns=null;continue;}
  if(inFence)continue;
  if(line.startsWith('|')){const n=line.split('|').length-2;if(columns===null){columns=n;tables++;}else if(n!==columns)errors.push({file,line:i+1,error:'Table columns'});}else columns=null;
 }
 const prose=source.replace(new RegExp(fence+'[^\\n]*\\n[\\s\\S]*?'+fence,'g'),'');
 for(const m of prose.matchAll(/\]\((<[^>]+>|[^)\n]+)\)/g)){
  const href=decodeURIComponent(m[1].replace(/^<|>$/g,'').split('#')[0]);if(!href||/^[a-z]+:\/\//i.test(href))continue;
  if(/^[a-z]:[/\\]/i.test(href)||!fs.existsSync(path.resolve(path.dirname(file),href)))errors.push({file,href,error:'Broken or absolute link'});
 }
 results.push({file:path.relative(root,file).replaceAll('\\','/'),display,inline,tables});
}
const result={passed:!errors.length,documents:results,display_formulas:results.reduce((a,x)=>a+x.display,0),inline_formulas:results.reduce((a,x)=>a+x.inline,0),tables:results.reduce((a,x)=>a+x.tables,0),errors};
fs.writeFileSync(path.join(stage,'analysis/repository_qa/markdown_math_checks.json'),JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify({...result,documents:results.length}));if(errors.length)process.exitCode=1;
