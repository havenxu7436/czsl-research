from pathlib import Path
import json,sys,numpy as np,torch
R=Path.cwd();src=R/'output/trustg0_cgqa_results_20261030_102311/trustg0_cgqa_results';out=R/'output/TRUST-G0_CGQA_20261030_analysis';ds=src/'work/cgqa';index=json.loads((ds/'index.json').read_text());sys.path.insert(0,str(src));from trustg0.utility import rank_corr
load=lambda p:torch.load(p,map_location='cpu',weights_only=False)
rows=[];noise=[];routes=[]
for seed in (5351,5352,5353):
 w=ds/f'seed{seed}'
 for f in (w/'recognition').glob('*/*/test_*.npz'):
  with np.load(f) as z:s=dict(z)
  seen=s['is_seen'].astype(bool)
  rows.append(dict(seed=seed,kind='formal',method=f.parent.parent.name,bank=int(f.parent.name),phase=f.stem,seen_nll=float(-s['truth_logprob'][seen].mean()),unseen_nll=float(-s['truth_logprob'][~seen].mean())))
 for t in (1,2,3):
  for f in (w/'q5_decision'/f'teacher{t}').glob('*.npz'):
   with np.load(f) as z:s=dict(z)
   seen=s['is_seen'].astype(bool)
   rows.append(dict(seed=seed,kind='q5',teacher=t,method=f.stem,phase='q5',seen_images=int(seen.sum()),unseen_images=int((~seen).sum()),seen_nll=float(-s['truth_logprob'][seen].mean()),unseen_nll=float(-s['truth_logprob'][~seen].mean())))
 for q in (1,2,3,4,5):
  for t in (1,2,3):
   r=load(w/'teachers'/f't{t}_q{q}'/'measure/response.pt');ef=r['effects'].astype(float);groups={}
   for j,i in enumerate(r['query_ids']):groups.setdefault(index['rows'][i]['pair'],[]).append(j)
   halves=[[v[c] for v in groups.values() if len(v)>=2] for c in (0,1)]
   yy=[ef[:,:,ix].mean((1,2)) for ix in halves]
   noise.append(dict(seed=seed,fold=q,teacher=t,query_images=len(r['query_ids']),pairs=len(groups),rho_split_half=rank_corr(*yy),target_rms=float(np.sqrt(np.mean(r['target']**2))),half_difference_rms=float(np.sqrt(np.mean((yy[0]-yy[1])**2)))))
   routes.append(dict(seed=seed,fold=q,teacher=t,mass=r['context_mass'].tolist()))
result={}
for kind in ('q5','formal'):
 for phase in (('q5',) if kind=='q5' else ('test_cw','test_ow')):
  for base in ('uniform','matched','context_perm_mean'):
   for metric in ('seen_nll','unseen_nll'):
    vals=[];individual=[]
    for seed in (5351,5352,5353):
     def v(m):return [r[metric] for r in rows if r['seed']==seed and r['kind']==kind and r['phase']==phase and (r['method'].startswith(m[:-5]) if m.endswith('_mean') else r['method']==m)]
     vals.append(float(np.mean(v('context'))-np.mean(v(base))))
     if kind=='q5':
      for t in (1,2,3):
       def vt(m):return [r[metric] for r in rows if r['seed']==seed and r['kind']==kind and r['teacher']==t and (r['method'].startswith(m[:-5]) if m.endswith('_mean') else r['method']==m)]
       individual.append(float(np.mean(vt('context'))-np.mean(vt(base))))
    result[kind+'_'+phase+'_'+base+'_'+metric]=dict(mean=float(np.mean(vals)),per_seed=vals,per_teacher=individual)
(out/'nll_subgroups.json').write_text(json.dumps(dict(rows=rows,paired=result),indent=2))
(out/'label_reliability.json').write_text(json.dumps(dict(noise=noise,context_mass=routes),indent=2))
print(json.dumps(result,indent=2))
print('NOISE')
for q in (1,2,3,4,5):
 z=[r for r in noise if r['fold']==q];print(q,'rho',np.mean([r['rho_split_half'] for r in z]),'range',min(r['rho_split_half'] for r in z),max(r['rho_split_half'] for r in z),'rms',np.mean([r['target_rms'] for r in z]),'halfdiff',np.mean([r['half_difference_rms'] for r in z]))
