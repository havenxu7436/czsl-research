from pathlib import Path
import json,csv,sys,hashlib
import numpy as np
import torch

ROOT=Path.cwd()
SOURCES={'UT':('trustg0_ut_results_20261030_083432/trustg0_ut_results','ut-zap50k'),
         'MIT':('trustg0_mit_results_20261030_085219/trustg0_mit_results','mit-states')}
OUT=ROOT/'output/TRUST-G0_UT_MIT_20261030_analysis';OUT.mkdir(exist_ok=True)
read=lambda p:json.loads(Path(p).read_text(encoding='utf-8-sig'))
load=lambda p:torch.load(p,map_location='cpu',weights_only=False)
sys.path.insert(0,str(ROOT/'output'/SOURCES['MIT'][0]));from trustg0 import utility
result={}
for name,(path,dsname) in SOURCES.items():
 src=ROOT/'output'/path;ds=src/'work'/dsname;out=ROOT/f'output/TRUST-G0_{name}_20261030_analysis'
 ix=read(ds/'index.json');bank=load(ds/'cache/bank.pt');types=bank['types'].numpy();seeds=[5351,5352,5353]
 audit=[];policy=[];splitchecks=[];bank_dif=[]
 recalc=read(out/'formal_recomputed.json');summary=list(csv.DictReader((src/'summary.csv').open(encoding='utf-8-sig',newline='')))
 print(name,'CSV_KEYS',summary[0].keys(),flush=True)
 # Independent full natural image-set and teacher training-split checks.
 for seed in seeds:
  w=ds/f'seed{seed}'
  for f in (w/'recognition').glob('*/*/metrics.json'):
   for phase in ('val_cw','val_ow','test_cw','test_ow'):
    with np.load(f.parent/(phase+'.npz')) as z:
     expected=[i for i,r in enumerate(ix['rows']) if r['phase']==phase.split('_')[0]]
     assert sorted(z['image_ids'].tolist())==expected
     train=set(ix['pairs']['train']);assert np.array_equal(z['is_seen'],[int(p) in train for p in z['truth']])
  for fold in read(w/'folds.json'):
   ids=fold['train_ids']+fold['audit_seen_ids']+fold['query_ids']
   assert all(ix['rows'][i]['phase']=='train' for i in ids)
  for stage in ('g0','uniform','capacity'):
   h=read(w/'formal'/stage/'history.json');receipt=read(w/'formal'/stage/'complete.json')
   best=max(h,key=lambda r:r['val']['AUC_native'])['epoch'];assert best==receipt['selected_epoch']
   splitchecks.append(dict(seed=seed,stage=stage,selected=best,val_best=max(r['val']['AUC_native'] for r in h),val_last=h[-1]['val']['AUC_native']))
  models=load(w/'utility.pt')
  for q in (4,5):
   for t in (1,2,3):
    r=load(w/'teachers'/f't{t}_q{q}'/'measure/response.pt');y=r['context_target'];mass=r['context_mass']
    def center(a):
     b=a.copy()
     for ty in range(3):b[:,types==ty]-=b[:,types==ty].mean(1,keepdims=True)
     return b
    for m in ('matched','context','context_shuffle'):
     p=np.broadcast_to(utility.predict(models[m],r),y.shape)
     rmse=float(np.sqrt(np.mean(np.sum(mass[:,None]*(p-y)**2,axis=0))))
     cy,cp=center(y),center(p)
     cr=float(np.sqrt(np.mean(np.sum(mass[:,None]*(cp-cy)**2,axis=0))))
     base=float(np.sqrt(np.mean(np.sum(mass[:,None]*cy**2,axis=0))))
     audit.append(dict(seed=seed,fold=q,teacher=t,method=m,conditional_rmse=rmse,centered_rmse=cr,
                      target_centered_rms=base,centered_skill=1-cr**2/base**2,
                      centered_rho=utility.rank_corr(cp.ravel(),cy.ravel()),context_mass=mass.tolist()))
  ps=load(w/'policies.pt')
  for m in ('matched','context','context_shuffle'):
   pi=ps[m].numpy()
   policy.append(dict(seed=seed,method=m,std=float(np.std(pi-.5)),min=float(pi.min()),max=float(pi.max()),
     mean_abs_deviation=float(np.abs(pi-.5).mean()),context_difference=float(np.abs(pi[0]-pi[-1]).mean())))
  for phase in ('test_cw','test_ow'):
   for ref in ('uniform','context_perm_mean'):
    differences=[]
    for bs in (1701,2701,3701):
     a=[r['AUC_native'] for r in recalc if r['seed']==seed and r['phase']==phase and r['method']=='context' and r['bank']==bs][0]
     b=[r['AUC_native'] for r in recalc if r['seed']==seed and r['phase']==phase and r['bank']==bs and (r['method'].startswith('context_perm') if ref.endswith('_mean') else r['method']==ref)]
     differences.append(float(a-np.mean(b)))
    bank_dif.append(dict(seed=seed,phase=phase,reference=ref,values=differences,mean=float(np.mean(differences)),sd=float(np.std(differences,ddof=1))))
 # Compare CSV to saved metrics independently, using its established field names.
 bykey={(r['seed'],r['method'],r['bank'],r['phase']):r for r in recalc}
 for r in summary:
  key=(int(r['seed']),r.get('method',r.get('variant')),int(r.get('bank',r.get('bank_seed'))),r.get('phase',r.get('world')))
  v=bykey[key]
  for k,value in r.items():
   if k in v and isinstance(v[k],(float,int)) and k not in ('seed','bank') and value!='':assert abs(float(value)-v[k])<1e-10
 v=read(out/'verification.json');v.update(summary_csv_rows_verified=len(summary),all_natural_phase_images_covered=True,teacher_uses_natural_train_only=True,formal_checkpoint_selection_matches_val_argmax=True)
 (out/'verification.json').write_text(json.dumps(v,indent=2),encoding='utf8')
 counts={phase:sum(r['phase']==phase for r in ix['rows']) for phase in ('train','val','test')}
 pairs={phase:len(ix['pairs'][phase]) for phase in ('train','val','test')}
 result[name]=dict(image_counts=counts,pair_counts=pairs,attrs=len(ix['attrs']),objs=len(ix['objs']),
  edges=len(types),edge_counts=np.bincount(types).tolist(),conditional_audits=audit,policies=policy,
  epoch_selection=splitchecks,bank_differences=bank_dif)
 print(name,'split',counts,pairs,'edges',np.bincount(types).tolist(),flush=True)
 for q in (4,5):
  for method in ('matched','context','context_shuffle'):
   rr=[r for r in audit if r['fold']==q and r['method']==method]
   print(name,'Q',q,method,{k:float(np.mean([r[k] for r in rr])) for k in ('conditional_rmse','centered_rmse','target_centered_rms','centered_skill','centered_rho')},flush=True)
 for phase in ('test_cw','test_ow'):
  for ref in ('uniform','context_perm_mean'):
   rr=[r for r in bank_dif if r['phase']==phase and r['reference']==ref]
   print(name,phase,ref,'bank_dif',[[round(v,5) for v in r['values']] for r in rr],flush=True)
(OUT/'joint_evidence.json').write_text(json.dumps(result,indent=2),encoding='utf8')
