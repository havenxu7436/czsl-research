def read(p): return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def sha(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for b in iter(lambda:f.read(4*1024*1024),b''): h.update(b)
    return h.hexdigest()
def write(name,value):
    def convert(v):
        if isinstance(v,np.ndarray):return v.tolist()
        if isinstance(v,np.generic):return v.item()
        if isinstance(v,Path):return str(v)
        raise TypeError(type(v))
    (OUT/name).write_text(json.dumps(value,default=convert,ensure_ascii=False,indent=2,allow_nan=False)+'\n',encoding='utf8')
def close(a,b,tol=1e-9):
    a=np.asarray(a);b=np.asarray(b)
    err=float(np.max(np.abs(a-b)))
    if err>tol or not np.isfinite(err):raise AssertionError((err,tol))
    return err
def stats(x):
    x=np.asarray(x,dtype=float)
    return dict(mean=float(x.mean()),sd=float(x.std(ddof=1)) if len(x)>1 else 0.,values=x.tolist())
