def f(x,n=3):return f'{float(x):.{n}f}'
def sign(x,n=3):return f'{float(x):+.{n}f}'
def table(headers,rows):
    return '\n'.join(['| '+' | '.join(headers)+' |','| '+' | '.join(['---']*len(headers))+' |']+['| '+' | '.join(map(str,r))+' |' for r in rows])
def val(method,world,metric='AUC_native',state='uniform_best',bank=1701,phase='test'):
    a=S[(S.policy==method)&(S.world==world)&(S.state==state)&(S.phase==phase)]
    if bank is not None:a=a[a.bank==bank]
    return a.groupby('seed')[metric].mean()
def paired(method,control,world,state='uniform_best',bank=None,phase='test',metric='AUC_native'):
    a=P[(P.comparison==method+'-minus-'+control)&(P.world==world)&(P.state==state)&(P.phase==phase)]
    if bank is not None:a=a[a.bank==bank]
    return a.groupby('seed')[metric].mean()
def qgain(method,control,metric='AUC_exact'):
    return QP[QP.comparison==method+'-minus-'+control].groupby('seed')[metric].mean()
def link(p,label):return '['+label+'](<'+Path(p).resolve().as_posix()+'>)'
