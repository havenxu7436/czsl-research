from pathlib import Path
import json,numpy as np,pandas as pd
OUT=Path(__file__).resolve().parent
W=OUT.parent/'output/state_results/server_results_20261025_113118/trustcpf_state_v1_10ep/work'
s=pd.read_csv(W/'summary.csv');p=pd.read_csv(W/'paired.csv');q=pd.read_csv(W/'q5_decisions.csv');qp=pd.read_csv(W/'q5_paired.csv')
si=s.set_index(['seed','state','policy','bank','phase','world']);qi=q.set_index(['seed','policy','bank']);err=0.
def check(a,b):
    global err
    e=abs(float(a)-float(b));err=max(err,e);assert e<1e-10,(a,b,e)
for row in p.itertuples():
    method,control=row.comparison.split('-minus-')
    names=[method+'_perm%d'%j for j in range(7101,7106)] if control=='permutation_mean' else ['pooled_absolute_cross_response'] if control=='cross_response' else [control]
    own=si.loc[(row.seed,row.state,method,row.bank,row.phase,row.world)]
    refs=pd.DataFrame([si.loc[(row.seed,row.state,n,row.bank,row.phase,row.world)] for n in names])
    for k in ['AUC_native','AUC_exact','HM']:check(getattr(row,k),own[k]-refs[k].mean())
    check(row.raw_NLL_gain,refs.raw_NLL.mean()-own.raw_NLL)
for row in qp.itertuples():
    method,control=row.comparison.split('-minus-')
    names=[method+'_perm%d'%j for j in range(7101,7106)] if control=='permutation_mean' else [control]
    own=qi.loc[(row.seed,method,row.bank)];refs=pd.DataFrame([qi.loc[(row.seed,n,row.bank)] for n in names])
    for k in ('AUC_exact','AUC_native'):check(getattr(row,k),own['mixture_'+k]-refs['mixture_'+k].mean())
    check(row.NLL_gain,refs.mixture_unll.mean()-own.mixture_unll)
agg=json.loads((W/'aggregates.json').read_text())
for a in agg:
    name=a['table'];table=s if name.startswith('recognition') else p if name.startswith('paired') else q
    keys=['state','policy' if name.startswith('recognition') else 'comparison','phase','world'] if name!='q5_bankmean' else ['policy']
    if not name.endswith('bankmean'):keys+=['bank']
    subset=table
    for k in keys:subset=subset[subset[k]==a[k]]
    for k in a:
        if not k.endswith('_mean'):continue
        metric=k[:-5];v=subset.groupby('seed')[metric].mean()
        assert len(v)==3
        check(a[k],v.mean());check(a[metric+'_sd'],v.std(ddof=1))
result=dict(passed=True,recognition_rows=len(s),paired_rows=len(p),q5_rows=len(q),q5_paired_rows=len(qp),aggregate_records=len(agg),max_error=err)
(OUT/'table_verification.json').write_text(json.dumps(result,indent=2),encoding='utf8')
print(json.dumps(result))
