def f(x,n=3):return f'{float(x):.{n}f}'
def sg(x,n=3):return f'{float(x):+.{n}f}'
def table(headers,rows):
 return '\n'.join(['| '+' | '.join(headers)+' |','| '+' | '.join(['---']*len(headers))+' |']+['| '+' | '.join(map(str,r))+' |' for r in rows])
def val(method,world,metric='AUC_native',bank=1701,phase='test'):
 a=S[(S.policy==method)&(S.world==world)&(S.phase==phase)]
 if bank is not None:a=a[a.bank==bank]
 return a.groupby('seed')[metric].mean()
def paired(method,control,world,bank=None,phase='test',metric='AUC_native'):
 a=P[(P.comparison==method+'-minus-'+control)&(P.world==world)&(P.phase==phase)]
 if bank is not None:a=a[a.bank==bank]
 return a.groupby('seed')[metric].mean()
def qgain(method,control='uniform',metric='AUC_exact'):
 return QP[QP.comparison==method+'-minus-'+control].groupby('seed')[metric].mean()
def link(p,label):return '['+label+'](<'+Path(p).resolve().as_posix()+'>)'
def avgpm(v):return f(v.mean())+' ± '+f(v.std(ddof=1))
def both(method,control,**kwargs):return ' / '.join(sg(paired(method,control,w,**kwargs).mean()) for w in ['cw','ow'])
