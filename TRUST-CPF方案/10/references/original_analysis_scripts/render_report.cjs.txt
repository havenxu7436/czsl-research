const fs=require('fs');
const katex=require('../tmp/katex/katex.min.cjs');
const {chromium}=require('C:/Users/llenovo/.cache/codex-runtimes/codex-primary-runtime/dependencies/node/node_modules/playwright');
const root=__dirname.replaceAll('\\','/');
fs.mkdirSync(root+'/browser-tmp',{recursive:true});process.env.TEMP=root+'/browser-tmp';process.env.TMP=root+'/browser-tmp';
const equations=JSON.parse(fs.readFileSync(root+'/math.json','utf8'));
let body=fs.readFileSync(root+'/body.html','utf8'),errors=[];
for(const [i,q] of equations.entries()){
 try{
  const html=katex.renderToString(q.tex,{displayMode:q.display,output:'mathml',throwOnError:true,strict:'ignore'});
  body=body.replace(q.display?'<p>'+q.token+'</p>':q.token,q.display?'<div class="mathblock" data-index="'+i+'">'+html+'</div>':'<span>'+html+'</span>');
 }catch(e){errors.push({i,error:e.message});}
}
if(errors.length)throw new Error(JSON.stringify(errors));
const css='body{max-width:1160px;margin:32px auto;padding:0 28px;font:16px/1.8 "Microsoft YaHei",sans-serif;color:#203348;background:white}h1{font-size:30px}h2{font-size:24px;margin-top:40px}table{width:100%;border-collapse:collapse;margin:18px 0;font-size:14px}th,td{padding:9px 11px;vertical-align:top;border-bottom:1px solid #ced9e2;text-align:left}th{background:#edf3f8}code{font:14px Consolas,monospace;overflow-wrap:anywhere;background:#f0f4f7}a{color:#1e5a8a;overflow-wrap:anywhere}.mathblock{font-family:"Cambria Math",serif;font-size:18px;padding:18px 0;margin:14px 0;background:#f7fafc}math{max-width:100%}img{max-width:100%;height:auto}';
fs.writeFileSync(root+'/report.html','<!doctype html><html lang="zh-CN"><meta charset="utf-8"><style>'+css+'</style><article>'+body+'</article></html>');
(async()=>{
 const browser=await chromium.launch({headless:true,executablePath:'C:/Program Files (x86)/Microsoft/Edge/Application/msedge.exe'});
 const page=await browser.newPage({viewport:{width:1440,height:1160},deviceScaleFactor:1});
 await page.goto('file:///'+root+'/report.html');await page.evaluate(()=>document.fonts.ready);
 const overflow=await page.locator('article,table,.mathblock').evaluateAll(ns=>ns.filter(n=>n.scrollWidth>n.clientWidth+3).map(n=>({tag:n.tagName,width:n.clientWidth,scroll:n.scrollWidth,text:n.innerText.slice(0,80)})));
 const missing=await page.locator('img').evaluateAll(ns=>ns.filter(n=>!n.complete||!n.naturalWidth).map(n=>n.src));
 await page.screenshot({path:root+'/preview_top.png'});
 for(const [name,heading] of [['main','2. 正式识别'],['q','4. Q4/Q5'],['next','7. 下一步']]){
  await page.getByRole('heading').filter({hasText:heading}).scrollIntoViewIfNeeded();
  await page.screenshot({path:root+'/preview_'+name+'.png'});
 }
 await page.locator('.mathblock').scrollIntoViewIfNeeded();await page.screenshot({path:root+'/preview_formula.png'});
 const result={passed:!overflow.length&&!missing.length,formulas:equations.length,tables:await page.locator('table').count(),images:await page.locator('img').count(),overflow,missing};
 fs.writeFileSync(root+'/layout_checks.json',JSON.stringify(result,null,2));console.log(JSON.stringify(result));await browser.close();if(!result.passed)process.exitCode=1;
})().catch(e=>{console.error(e);process.exitCode=1});
