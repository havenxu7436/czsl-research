from pathlib import Path
import json,sys
import numpy as np
import pandas as pd
sys.dont_write_bytecode=True
OUT=Path(__file__).resolve().parent
ROOT=OUT.parent
BASE=ROOT/'output/abc_results/server_results_20261027_075618/trustcpf_abc_v1_10ep'
W=BASE/'work'
s=pd.read_csv(W/'summary.csv')
rows=[];maxerr=0.
for seed in (3351,3352,3353):
    w=W/f'seed{seed}'
    for bank in (1701,2701,3701):
        for policy in ('uniform','matched','amplitude','mix16','context','context_shuffle'):
            ref=json.loads((w/f'recognition/bank{bank}/policies/{policy}.json').read_text())
            with np.load(w/ref['result_directory']/'test_predictions.npz') as z:
                routes=z['routes']
                for world in ('cw','ow'):
                    x=z[world+'_logprob'].astype(float);y=z[world+'_truth'];seen=z[world+'_seen']
                    si=np.flatnonzero(seen);ui=np.flatnonzero(~seen)
                    sp=si[x[:,si].argmax(1)];up=ui[x[:,ui].argmax(1)]
                    d=x[np.arange(len(x)),sp]-x[np.arange(len(x)),up]
                    sm=seen[y];um=~sm;cs=sm&(sp==y);cu=um&(up==y)
                    total=0.
                    for g in (0,1):
                        for h in (0,1):
                            sv=d[cs&(routes==g)];uv=np.sort(d[cu&(routes==h)])
                            left=np.searchsorted(uv,sv,side='left');right=np.searchsorted(uv,sv,side='right')
                            value=(left+(right-left)*.5).sum()/sm.sum()/um.sum()*100
                            total+=value
                            rows.append(dict(seed=seed,bank=bank,policy=policy,world=world,seen_group=g,unseen_group=h,AUC_contribution=value))
                    target=float(s[(s.seed==seed)&(s.bank==bank)&(s.policy==policy)&(s.phase=='test')&(s.world==world)].iloc[0].AUC_exact)
                    maxerr=max(maxerr,abs(target-total))
                    assert abs(total-target)<1e-10
parts=pd.DataFrame(rows)
parts.to_csv(OUT/'auc_context_decomposition.csv',index=False)
by=parts.groupby(['seed','policy','world','seen_group','unseen_group']).AUC_contribution.mean().reset_index()
gains=[]
for control in ('uniform','matched','context_shuffle'):
    a=by[by.policy=='context'].drop(columns='policy')
    b=by[by.policy==control].drop(columns='policy')
    merged=a.merge(b,on=['seed','world','seen_group','unseen_group'],suffixes=('_c','_b'))
    merged['gain']=merged.AUC_contribution_c-merged.AUC_contribution_b
    merged['comparison']='context-minus-'+control
    gains.append(merged)
gain=pd.concat(gains,ignore_index=True)
gain.to_csv(OUT/'auc_context_decomposition_paired.csv',index=False)

old=ROOT/'output/stage_results/server_results_20261025_183634/trustcpf_stage_v1_10ep/work'
old_s=pd.read_csv(old/'summary.csv')
prior=old_s[(old_s.phase=='test')&old_s.permutation_seed.isna()].groupby(['policy','world'])[['AUC_native','AUC_exact']].mean().reset_index()
print('PRIOR',prior.to_string(index=False))
qpaired={}
for fold in (4,5):
    q=pd.read_csv(W/f'q{fold}_curves.csv');p=pd.read_csv(W/f'q{fold}_paired.csv');error=0.
    for r in p.to_dict('records'):
        method,control=r['comparison'].split('-minus-')
        a=q[(q.seed==r['seed'])&(q.bank==r['bank'])&(q.policy==method)].iloc[0]
        table=q[(q.seed==r['seed'])&(q.bank==r['bank'])]
        if control=='permutation_mean':b=table[table.policy.str.startswith(method+'_perm')]
        elif control=='matched':
            key='matched_e%03d'%round(a.eta*100)
            b=table[table.policy==key]
        else:b=table[table.policy==control]
        assert len(b)>0,(method,control)
        for col in ('mixture_AUC_exact','mixture_unll','single_unll'):
            e=abs(float(a[col]-b[col].mean())-r[col]);error=max(error,e)
            assert e<1e-10,(r,col,e)
    qpaired[str(fold)]=dict(rows=len(p),max_error=error)

out=dict(pair_auc_max_error=maxerr,prior_formal=prior.to_dict('records'),q_paired=qpaired,
         contribution_by_seed=by.to_dict('records'),contribution_gains=gain.to_dict('records'))
(OUT/'supplement.json').write_text(json.dumps(out,ensure_ascii=False,indent=2)+'\n',encoding='utf8')
print('PAIR_DECOMPOSITION maxerr',maxerr)
print(gain.groupby(['comparison','world','seen_group','unseen_group']).gain.mean().to_string())
print('QPAIRED',qpaired)
