def stats(x):
 a=np.asarray(x,float)
 return dict(mean=float(a.mean()),sd=float(a.std(ddof=1)) if len(a)>1 else None,values=a.tolist())

def array_corr(a,b):
 a=np.asarray(a,float).ravel();b=np.asarray(b,float).ravel();a=a-a.mean();b=b-b.mean()
 return float(a@b/np.sqrt((a@a)*(b@b))) if a@a>0 and b@b>0 else None

def compare_numbers(a,b):
 errors=[]
 if isinstance(a,dict):
  for k in b:
   if k in a:errors.extend(compare_numbers(a[k],b[k]))
 elif isinstance(a,(int,float)) and isinstance(b,(int,float)):errors.append(abs(a-b))
 return errors

def audit_archive():
 archive=ROOT/'trustcpf_conditional_v2_results_20261022_232124.tar.gz'
 expected=Path(str(archive)+'.sha256').read_text().split()[0]
 actual=sha(archive);assert expected==actual
 missing=[];different=[];count=0
 with tarfile.open(archive,'r:gz') as tar:
  for m in tar:
   if not m.isfile():continue
   count+=1;p=RUN.parent/m.name
   h=hashlib.sha256()
   with tar.extractfile(m) as f:
    for chunk in iter(lambda:f.read(4*1024*1024),b''):h.update(chunk)
   if not p.exists():missing.append(m.name)
   elif sha(p)!=h.hexdigest():different.append(m.name)
 release=read(RUN/'MANIFEST.json')
 source_changes=[n for n,h in release.items() if not (RUN/n).exists() or sha(RUN/n)!=h]
 assert not missing and not different and not source_changes,(missing,different,source_changes)
 return dict(compressed_sha256=actual,archive_regular_files=count,missing=missing,content_differences=different,
             release_files=len(release),release_differences=source_changes)

 print('PREDICTIONS_RECOMPUTED',maxerror,flush=True)
